## Method and false-positive risk
Parses the spec with the repo's yaml package; extracts @Controller + @Get/@Post/@Put/@Patch/@Delete by regex, prefixes the global prefix v1 (app.setup.ts, setGlobalPrefix; no URI versioning), normalises params to :p. No non-decorator route registration exists (grep for adapter/express registration found none); webhooks use decorators. Validated on seller analytics (`/v1/seller/stores/:p/analytics/*` found) and GET /v1/admin/ledger/balances (found, also in ledger http test). Risks: regex does not resolve controller paths built from constants (0 dynamic prefixes detected); a literal-vs-param route collision is not flagged; method/path equality means a route can be served with a different contract; test-only controllers excluded by filename. Caller detection greps path strings (method not checked, so PATCH /v1/me hits are probably GET /me) across apps/mobile, admin-web, seller-web, shopper-web, packages, tooling; "api-client (generated)" means only the generated client (built from the spec) references it, i.e. no real consumer.

## Appendix A: diff-routes.mjs (output to out.json)
```js
// Usage: node diff-routes.mjs /Users/nadakhalifa/OUTLET   (needs repo node_modules/yaml; read-only)
import { readFileSync, readdirSync, statSync } from 'node:fs';
import { join, relative } from 'node:path';
import { createRequire } from 'node:module';
const root = process.argv[2] ?? '/Users/nadakhalifa/OUTLET';
const { parse } = createRequire(join(root, 'package.json'))('yaml');
const M = ['get', 'put', 'post', 'delete', 'patch'];
const norm = (p) => '/' + p.replace(/\{[^}]+\}|:[A-Za-z_][A-Za-z0-9_]*/g, ':p').split('/').filter(Boolean).join('/');
const spec = parse(readFileSync(join(root, 'docs/api/openapi.yaml'), 'utf8'));
const ops = [];
for (const [p, item] of Object.entries(spec.paths ?? {}))
  for (const m of M) if (item[m]) ops.push({ method: m.toUpperCase(), path: p, key: `${m.toUpperCase()} ${norm(p)}`, tag: (item[m].tags ?? [])[0] ?? '' });
const walk = (d) => readdirSync(d).flatMap((f) => { const q = join(d, f); return statSync(q).isDirectory() ? (f === 'node_modules' ? [] : walk(q)) : [q]; });
const src = join(root, 'apps/api/src');
const files = walk(src).filter((f) => f.endsWith('.ts') && !/\.test\.ts$|\/test\/|\.spec\.ts$/.test(f));
const routes = [];
const lit = (s) => { const m = s.match(/['"`]([^'"`]*)['"`]/); return m ? m[1] : ''; };
for (const f of files) {
  const t = readFileSync(f, 'utf8');
  if (!t.includes('@Controller(')) continue;
  const rel = relative(src, f);
  const mod = rel.startsWith('modules/') ? rel.split('/')[1] : rel.split('/')[0];
  const parts = t.split(/(?=@Controller\()/).slice(1);
  for (const part of parts) {
    const cm = part.match(/^@Controller\(([^)]*)\)/);
    let prefix = cm[1].trim().startsWith('{') ? (cm[1].match(/path:\s*['"`]([^'"`]*)/) ?? [, ''])[1] : lit(cm[1]);
    if (/^\s*[A-Za-z_]/.test(cm[1]) && !cm[1].includes('"') && !cm[1].includes("'")) prefix = '<<DYNAMIC:' + cm[1] + '>>';
    const body = part.split(/\nexport class|\nclass /)[0] === part ? part : part;
    const re = /@(Get|Post|Put|Patch|Delete)\(([^)]*)\)/g; let m;
    while ((m = re.exec(body))) routes.push({ method: m[1].toUpperCase(), path: '/v1/' + [prefix, lit(m[2])].filter(Boolean).join('/'), file: rel, mod });
  }
}
for (const r of routes) r.key = `${r.method} ${norm(r.path)}`;
const rk = new Set(routes.map((r) => r.key)), ok = new Set(ops.map((o) => o.key));
const unserved = ops.filter((o) => !rk.has(o.key));
const extra = routes.filter((r) => !ok.has(r.key));
console.log(JSON.stringify({ specOps: ops.length, controllerRoutes: routes.length, served: ops.length - unserved.length, unserved, extra, dynamic: routes.filter((r) => r.path.includes('DYNAMIC')), routes }, null, 1));
```

## Appendix B: callers.mjs (reads out.json)
```js
// node callers.mjs <root> : for each unserved op, grep client code for the path (params -> wildcard)
import { readFileSync, readdirSync, statSync } from 'node:fs';
import { join, relative } from 'node:path';
const root = process.argv[2];
const j = JSON.parse(readFileSync(new URL('./out.json', import.meta.url)));
const walk = (d) => readdirSync(d).flatMap((f) => { const q = join(d, f); if (['node_modules', '.next', 'dist', '.expo'].includes(f)) return []; return statSync(q).isDirectory() ? walk(q) : [q]; });
const dirs = ['apps/mobile', 'apps/admin-web', 'apps/seller-web', 'apps/shopper-web', 'packages', 'tooling'].map((d) => join(root, d));
const files = dirs.flatMap((d) => { try { return walk(d); } catch { return []; } }).filter((f) => /\.(ts|tsx|mjs|js)$/.test(f) && !/\.test\.|\.d\.ts/.test(f));
const texts = files.map((f) => [relative(root, f), readFileSync(f, 'utf8')]);
const out = {};
for (const o of j.unserved) {
  const tail = o.path.replace(/^\/v1/, '');
  const re = new RegExp(tail.replace(/[.*+?^$|()[\]\\]/g, '\\$&').replace(/\\?\{[^}]+\}/g, '[^\'"`/]*').replace(/\//g, '\\/') + '(?![A-Za-z0-9_-])');
  const hits = texts.filter(([f, t]) => re.test(t)).map(([f]) => f);
  const area = (f) => f.startsWith('apps/') ? f.split('/')[1] : f.startsWith('packages/') ? f.split('/').slice(0, 2).join('/') : f.split('/').slice(0, 2).join('/');
  out[o.method + ' ' + o.path] = [...new Set(hits.map(area))];
}
console.log(JSON.stringify(out, null, 1));
```
